二、PostgreSQL 迁移方案
2.1 为什么选 PostgreSQL
维度	SQLite	PostgreSQL
并发写入	写锁，单线程	MVCC，多连接并发
复杂查询	有限	窗口函数、CTE、JSONB
事务一致性	弱	强 ACID
CRM漏斗分析	吃力	游刃有余
未来扩展	无法水平扩展	支持读写分离、分库分表
数组/JSONB字段	无	原生支持
全文搜索	插件	内置
2.2 环境变量配置变更
现有的 database.config.ts 需要从 SQLite 切换为 PostgreSQL，只需改配置中的连接参数：
# .env 文件变更# 旧（SQLite）DB_DATABASE=./data/enbon-admin.db# 或DB_PATH=./data/enbon-admin.db# 新（PostgreSQL）DB_HOST=localhostDB_PORT=5432DB_USERNAME=postgresDB_PASSWORD=your_passwordDB_DATABASE=enbon_adminDB_SYNCHRONIZE=false    # 生产环境必须为 falseDB_LOGGING=true         # 开发环境开启 SQL 日志

2.5 需要修改的 Entity 字段（SQLite → PostgreSQL 兼容性调整）
TypeORM 的大多数字段 PostgreSQL 都兼容，只需要关注以下几处：
2.5.1 SQLite boolean → PostgreSQL 保持 boolean
// ❌ SQLite下有些用 0/1 存储的布尔值，PostgreSQL 用原生 boolean@Column({ type: 'boolean', default: false })enabled: boolean;// ✅ 建议统一显式指定 type 为 boolean@Column({ type: 'boolean', default: false })isActive: boolean;
2.5.2 JSON 字段（audit-log.entity.ts 中的 details 字段）
// PostgreSQL 建议用 jsonb（可索引的二进制 JSON），性能更好@Column({ type: 'text', nullable: true })  // SQLite@Column({ type: 'jsonb', nullable: true })  // PostgreSQL（索引友好）details: Record<string, any>;
2.5.3 加密字段（user.entity.ts 中的密码字段）
// AES-256-GCM 加密的字段用 text 存储，两端兼容@Column({ type: 'text', nullable: true })_facebookPassword: string;
2.5.4 User Entity 中的枚举字段
当前 User 实体大量使用 TypeScript 枚举，数据库端存储为 string。PostgreSQL 建议改用 enum 类型：
// 新建 enums 目录存放所有枚举export enum UserRole {  SUPER_ADMIN = 'SUPER_ADMIN',  DEPARTMENT_HEAD = 'DEPARTMENT_HEAD',  EMPLOYEE = 'EMPLOYEE',  HR_DIRECTOR = 'HR_DIRECTOR',  HR_RECEPTION = 'HR_RECEPTION',  FINANCE = 'FINANCE',  GUEST = 'GUEST',}// Entity 中使用@Column({ type: 'enum', enum: UserRole, default: UserRole.EMPLOYEE })role: UserRole;
> 注意：如果现有数据已经用 string 存储了枚举值，迁移脚本需要先创建 enum 类型，再 ALTER COLUMN。

// scripts/migrate-sqlite-to-pg.js 示例const sqlite3 = require('better-sqlite3');const { Client } = require('pg');async function migrate() {  const sqlite = sqlite3('./data/enbon-admin.db');  const pg = new Client({ connectionString: process.env.DATABASE_URL });  await pg.connect();  const tables = await sqlite.prepare(    "SELECT name FROM sqlite_master WHERE type='table'"  ).all();  for (const { name } of tables) {    if (name.startsWith('sqlite_')) continue; // 跳过系统表    const rows = sqlite.prepare(`SELECT * FROM ${name}`).all();    if (rows.length === 0) continue;    // 清洗数据    const cleanedRows = rows.map(row => {      // 1. 将 SQLite 的 0/1 布尔值转为 true/false      // 2. 将日期字符串转为 ISO 格式      // 3. 处理枚举值映射      return cleanRow(row);    });    // 批量插入    const columns = Object.keys(cleanedRows[0]);    const values = cleanedRows.map((r, i) =>      `(${columns.map((_, j) => `$${i * columns.length + j + 1}`).join(',')})`    ).join(',');    const query = `INSERT INTO ${name} (${columns.join(',')}) VALUES ${values} ON CONFLICT DO NOTHING`;    await pg.query(query, cleanedRows.flatMap(Object.values));    console.log(`Migrated ${rows.length} rows from ${name}`);  }  await pg.end();  sqlite.close();}

// backend/src/common/storage/minio.service.tsimport { Injectable, OnModuleInit } from '@nestjs/common';import * as Minio from 'minio';@Injectable()export class MinioService implements OnModuleInit {  private client: Minio.Client;    // 确保 bucket 存在    const exists = await this.client.bucketExists(this.bucket);    if (!exists) {      await this.client.makeBucket(this.bucket);    }  }  // 上传文件  async upload(    storageKey: string,    buffer: Buffer,    mimeType: string,    meta?: Record<string, string>,  ): Promise<string> {    await this.client.putObject(      this.bucket,      storageKey,      buffer,      buffer.length,      { 'Content-Type': mimeType, ...meta },    );    return storageKey;  }  // 下载文件  async download(storageKey: string): Promise<Buffer> {    const stream = await this.client.getObject(this.bucket, storageKey);    return this.streamToBuffer(stream);  }  // 生成预签名 URL（临时访问链接）  async getPresignedUrl(storageKey: string, expiresSec = 3600): Promise<string> {    return this.client.presignedGetObject(this.bucket, storageKey, expiresSec);  }  // 删除文件  async delete(storageKey: string): Promise<void> {    await this.client.removeObject(this.bucket, storageKey);  }  // 检查文件是否存在  async exists(storageKey: string): Promise<boolean> {    try {      await this.client.statObject(this.bucket, storageKey);      return true;    } catch {      return false;    }  }  private streamToBuffer(stream: NodeJS.ReadableStream): Promise<Buffer> {    return new Promise((resolve, reject) => {      const chunks: Buffer[] = [];      stream.on('data', (chunk) => chunks.push(Buffer.from(chunk)));      stream.on('end', () => resolve(Buffer.concat(chunks)));      stream.on('error', reject);    });  }}

// backend/src/modules/files/file-storage.service.ts@Injectable()export class FileStorageService {  constructor(    @InjectRepository(FileRecord)    private fileRecordRepo: Repository<FileRecord>,    @InjectRepository(FileShare)    private fileShareRepo: Repository<FileShare>,    @InjectRepository(FileAccessLog)    private fileAccessLogRepo: Repository<FileAccessLog>,    @InjectRepository(StorageBucket)    private bucketRepo: Repository<StorageBucket>,    private minioService: MinioService,    private auditService: AuditService,  ) {}  /**   * 上传文件（核心方法）   */  async uploadFile(    file: Express.Multer.File,    bucket: string,    ownerId: number,    options: {      module?: string;      moduleId?: string;      isPublic?: boolean;      allowedRoles?: string[];      allowedDepts?: string[];      overwriteHash?: string; // 传入已有文件的 hash 进行去重    },  ): Promise<FileRecord> {    // 1. 查找 bucket 配置    const bucketConfig = await this.bucketRepo.findOne({ where: { bucket_code: bucket } });    if (!bucketConfig || !bucketConfig.is_active) {      throw new BadRequestException(`存储桶 ${bucket} 不存在或已禁用`);    }    // 2. 检查文件大小    if (file.size > bucketConfig.max_file_size) {      throw new BadRequestException(`文件大小超过限制（${bucketConfig.max_file_size / 1024 / 1024}MB）`);    }    // 3. 检查扩展名    if (bucketConfig.allowed_extensions?.length > 0) {      const ext = path.extname(file.originalname).toLowerCase();      if (!bucketConfig.allowed_extensions.includes(ext)) {        throw new BadRequestException(`不允许的文件类型：${ext}`);      }    }    // 4. 计算文件哈希（用于去重）    const fileHash = crypto.createHash('sha256').update(file.buffer).digest('hex');    // 5. 检查是否已有相同文件（去重）    if (options.overwriteHash) {      const existing = await this.fileRecordRepo.findOne({        where: { file_hash: fileHash, bucket, is_deleted: false },      });      if (existing) {        return existing; // 返回已有记录，不重复上传      }    }    // 6. 生成存储路径：bucket/YYYY/MM/uuid.ext    const date = new Date();    const ext = path.extname(file.originalname);    const storageKey = `${bucket}/${date.getFullYear()}/${String(date.getMonth() + 1).padStart(2, '0')}/${uuidv4()}${ext}`;
